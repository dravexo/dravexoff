<script>
let score = 0;
let hits = 0;
let shots = 0;

let gameTime = 60;
let timerInterval = null;

let moveTimer = null;
let sensitivity = 900;

const dummy = document.getElementById("dummy");
const gameArea = document.getElementById("gameArea");
const scoreEl = document.getElementById("score");
const hsEl = document.getElementById("hs");
const timerEl = document.getElementById("timer");
const hsText = document.getElementById("headshotText");

/* MOVE DUMMY */
function moveDummy(){
  const maxX = gameArea.clientWidth - 100;
  const x = Math.random() * maxX;
  dummy.style.left = x + "px";
}

/* SET TIME */
function setTime(t){
  gameTime = t;
  timerEl.innerText = gameTime;
}

/* START */
function startGame(){
  if(moveTimer) return;

  score = hits = shots = 0;
  scoreEl.innerText = 0;
  hsEl.innerText = "0%";

  moveDummy();
  moveTimer = setInterval(moveDummy, sensitivity);
  startTimer();
}

/* TIMER */
function startTimer(){
  clearInterval(timerInterval);
  timerEl.innerText = gameTime;

  timerInterval = setInterval(()=>{
    gameTime--;
    timerEl.innerText = gameTime;

    if(gameTime <= 0){
      clearInterval(timerInterval);
      clearInterval(moveTimer);
      moveTimer = null;
      alert("TIME OVER!\nScore: " + score);
    }
  },1000);
}

/* RESET */
function resetGame(){
  clearInterval(timerInterval);
  clearInterval(moveTimer);
  moveTimer = null;

  score = hits = shots = 0;
  scoreEl.innerText = 0;
  hsEl.innerText = "0%";
  timerEl.innerText = gameTime;
}

/* SENSITIVITY */
function setSensitivity(val){
  sensitivity = val;
  if(moveTimer){
    clearInterval(moveTimer);
    moveTimer = setInterval(moveDummy, sensitivity);
  }
}

/* TOUCH AIM */
gameArea.addEventListener("touchstart", function(e){
  if(!moveTimer) return;

  const touch = e.touches[0];
  shots++;

  const headBox = document.querySelector(".head").getBoundingClientRect();
  const bodyBox = document.querySelector(".body").getBoundingClientRect();

  if(
    touch.clientX >= headBox.left &&
    touch.clientX <= headBox.right &&
    touch.clientY >= headBox.top &&
    touch.clientY <= headBox.bottom
  ){
    score += 10;
    hits++;
    showHeadshot();
    if(navigator.vibrate){ navigator.vibrate(80); }
  }
  else if(
    touch.clientX >= bodyBox.left &&
    touch.clientX <= bodyBox.right &&
    touch.clientY >= bodyBox.top &&
    touch.clientY <= bodyBox.bottom
  ){
    score += 3;
    hits++;
  }

  scoreEl.innerText = score;
  hsEl.innerText = Math.round((hits/shots)*100) + "%";
});

/* HEADSHOT EFFECT */
function showHeadshot(){
  hsText.style.opacity = 1;
  setTimeout(()=>{ hsText.style.opacity = 0; },300);
}
